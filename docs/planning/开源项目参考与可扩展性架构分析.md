```python
class QuotaManager:
    LIMITS = {

            "ai_calls_per_month": 100,

            "storage_mb": 100,
        },
        SubscriptionTier.PRO: {
            "ai_calls_per_month": 5000,

    async def check_quota(
        self,
        tenant_id: str,
        resource_type: str
    ) -> bool:
        tenant = await self.get_tenant(tenant_id)
        usage = await self.get_usage(tenant_id, resource_type)
        limit = self.LIMITS[tenant.tier][resource_type]

        if limit == -1:  # 无限制
            return True

        return usage < limit

    async def consume_quota(
        self,
        tenant_id: str,
        resource_type: str,
        amount: int = 1
    ):
        if not await self.check_quota(tenant_id, resource_type):
            raise QuotaExceededException(
                f"Quota exceeded for {resource_type}"
            )

        await self.increment_usage(tenant_id, resource_type, amount)

# FastAPI中间件
class TenantIsolationMiddleware:
    async def __call__(self, request: Request, call_next):
        # 从JWT或Header提取租户ID
        tenant_id = self._extract_tenant_id(request)

        if not tenant_id:
            raise HTTPException(status_code=401, detail="Tenant ID required")

        # 将租户ID注入到请求上下文
        request.state.tenant_id = tenant_id

        # 检查租户状态
        tenant = await self.tenant_service.get_tenant(tenant_id)
```

```python
from langchain.chains import GraphCypherQAChain
from langchain_community.graphs import Neo4jGraph
from langchain_openai import ChatOpenAI
from langchain.prompts import PromptTemplate

class GraphRAGSystem:
    def __init__(self):
        # Neo4j连接
        self.graph = Neo4jGraph(
            url="bolt://localhost:7687",
            username="neo4j",
            password="password"
        )

        # LLM
        self.llm = ChatOpenAI(temperature=0, model="gpt-4")

        # Cypher生成链
        self.cypher_chain = GraphCypherQAChain.from_llm(
            llm=self.llm,
            graph=self.graph,
            verbose=True
        )

    async def hybrid_query(
        self,
        query: str,
        use_vector: bool = True,
        use_graph: bool = True
    ) -> Dict:
        results = {}

        # 向量检索（语义相似）
        if use_vector:
            vector_results = await self._vector_search(query)
            results["vector"] = vector_results

        # 图遍历（关系推理）
        if use_graph:
            graph_results = await self._graph_traverse(query)
            results["graph"] = graph_results

        # 结果融合
        merged = self._merge_results(results)

        # LLM生成最终答案
        answer = await self._generate_answer(query, merged)

        return {
            "answer": answer,
            "sources": merged,
            "reasoning": self._explain_reasoning(merged)
        }

    async def _vector_search(self, query: str) -> List[Dict]:
        """向量检索：查找语义相似的文档"""
        cypher = """
        CALL db.index.vector.queryNodes(
            'document_embeddings',
            $k,
            $embedding
        ) YIELD node, score
        RETURN node.text as text, node.metadata as metadata, score
        ORDER BY score DESC
        """

        # 生成查询向量
        embedding = await self._get_embedding(query)

        result = self.graph.query(
            cypher,
            params={"k": 5, "embedding": embedding}
        )
        return result

    async def _graph_traverse(self, query: str) -> List[Dict]:
        """图遍历：基于关系推理"""
        # LLM生成Cypher查询
        cypher_query = self.cypher_chain.generate_cypher(query)

        # 执行图查询
        result = self.graph.query(cypher_query)
        return result

    def _merge_results(self, results: Dict) -> List[Dict]:
        """融合向量和图检索结果"""
        merged = []

        # 来自向量检索的文档
        for doc in results.get("vector", []):
            merged.append({
                "content": doc["text"],
                "source": "vector_search",
                "score": doc["score"],
                "type": "semantic_similarity"
            })

        # 来自图遍历的关系
        for item in results.get("graph", []):
            merged.append({
                "content": item,
                "source": "graph_traversal",
                "type": "relationship"
            })

        # 去重和排序
        return self._deduplicate_and_rank(merged)

# 农畜产品应用示例
class ProductGraphRAG(GraphRAGSystem):
    async def query_product_culture(self, product_name: str) -> Dict:
        """查询产品的文化背景"""
        query = f"{product_name}的文化背景和历史故事"

        # 图遍历：产品 -> 地域 -> 文化 -> 故事
        cypher = """
        MATCH (p:Product {name: $product_name})
              -[:ORIGIN_FROM]->(region:Region)
              -[:HAS_CULTURE]->(culture:Culture)
              -[:HAS_STORY]->(story:Story)
        RETURN p, region, culture, story
        """

        graph_results = self.graph.query(
            cypher,
            params={"product_name": product_name}
        )

        # 向量检索：查找相关文化资料
        vector_results = await self._vector_search(query)

        # 融合并生成营销文案
        return await self._generate_marketing_copy(
            product_name,
            graph_results,
            vector_results
        )
```

```python
class Text2CypherConverter:
    CYPHER_GENERATION_TEMPLATE = """
    基于Neo4j图谱Schema，将用户问题转换为Cypher查询。

    Schema:
    {schema}

    示例:
    问题: "哪些产品来自锡林郭勒盟？"
    Cypher: MATCH (p:Product)-[:ORIGIN_FROM]->(r:Region {{name: '锡林郭勒盟'}})
            RETURN p.name, p.description

    问题: "羊绒产品的文化故事有哪些？"
    Cypher: MATCH (p:Product {{category: '羊绒'}})-[:HAS_STORY]->(s:Story)
            RETURN s.title, s.content

    问题: {question}
    Cypher:
    """

    def __init__(self, graph: Neo4jGraph, llm: BaseLLM):
        self.graph = graph
        self.llm = llm
        self.schema = graph.get_schema

    async def convert(self, question: str) -> str:
        """将自然语言转为Cypher"""
        prompt = PromptTemplate(
            template=self.CYPHER_GENERATION_TEMPLATE,
            input_variables=["schema", "question"]
        )
```

```python
class Plugin(ABC):
    """插件基类"""

        pass

        self,
        inputs: Dict[str, Any],
        context: PluginContext
    ) -> Dict[str, Any]:
        """执行插件"""
        pass

    @abstractmethod
    def get_schema(self) -> Dict:
        """返回插件输入输出Schema"""
```

```python
    def register_provider(self, name: str, provider: LLMProvider):
        self._providers[name] = provider

    def get_provider(self, name: str) -> LLMProvider:
        if name not in self._providers:
            raise ValueError(f"Provider {name} not found")
        return self._providers[name]

    async def chat(
        self,
        provider_name: str,
        messages: List[Dict],
        **kwargs
    ):
        provider = self.get_provider(provider_name)
        return await provider.chat(messages, **kwargs)

# 使用示例：轻松切换LLM提供商
```

```python
# 工作流节点抽象
class WorkflowNode(ABC):
    @abstractmethod
    async def execute(
        self,
        inputs: Dict,
        context: WorkflowContext
    ) -> Dict:
        pass

# 具体节点示例
class LLMNode(WorkflowNode):
    """LLM生成节点"""
    async def execute(self, inputs: Dict, context: WorkflowContext):
        prompt = inputs["prompt"]
        llm = context.get_llm()
        response = await llm.chat([
            {"role": "user", "content": prompt}
        ])
        return {"output": response}

class GraphQueryNode(WorkflowNode):
    """知识图谱查询节点"""
    async def execute(self, inputs: Dict, context: WorkflowContext):
        product_name = inputs["product_name"]
        graph = context.get_graph_db()
        result = await graph.query(
            "MATCH (p:Product {name: $name})-[:HAS_CULTURE]->(c) RETURN c",
            {"name": product_name}
        )
        return {"culture": result}

# 工作流定义
class Workflow:
    def __init__(self, name: str):
        self.name = name
        self.nodes: List[WorkflowNode] = []
        self.edges: List[Tuple[int, int]] = []  # (from_node, to_node)

    def add_node(self, node: WorkflowNode) -> int:
        self.nodes.append(node)
        return len(self.nodes) - 1

    def add_edge(self, from_node: int, to_node: int):
        self.edges.append((from_node, to_node))

    async def execute(self, initial_inputs: Dict) -> Dict:
        """执行工作流"""
        # 拓扑排序
        sorted_nodes = self._topological_sort()

        # 执行节点
        context = WorkflowContext()
        data = initial_inputs

        for node_idx in sorted_nodes:
            node = self.nodes[node_idx]
            data = await node.execute(data, context)

        return data

# 使用示例：构建营销文案生成工作流
workflow = Workflow("product_marketing_copywriting")

# 1. 查询产品信息
product_node = workflow.add_node(ProductQueryNode())

# 2. 查询文化背景
culture_node = workflow.add_node(GraphQueryNode())

# 3. 生成文案
copywriting_node = workflow.add_node(LLMNode())

# 4. 平台适配
adapt_node = workflow.add_node(PlatformAdaptNode())

# 连接节点
workflow.add_edge(product_node, culture_node)
workflow.add_edge(culture_node, copywriting_node)
workflow.add_edge(copywriting_node, adapt_node)

# 执行工作流
result = await workflow.execute({
    "product_name": "锡林郭勒羊绒",
    "platform": "wechat"
})
```

```python
from crewai.flow.flow import Flow, listen, start

class MarketingContentFlow(Flow):
    """营销内容生成流程"""

    @start()
    def receive_request(self):
        """接收用户请求"""
        return {
            "product_name": self.state.product_name,
            "platform": self.state.platform
        }

    @listen(receive_request)
    def research_product(self, request):
        """研究产品"""
        crew = Crew(
            agents=[self.researcher],
            tasks=[self.research_task]
        )
        result = crew.kickoff(inputs=request)
        return result

    @listen(research_product)
    def generate_copywriting(self, research_result):
        """生成文案"""
        crew = Crew(
            agents=[self.copywriter],
            tasks=[self.copywriting_task]
        )
        result = crew.kickoff(inputs={
            "research": research_result,
            "platform": self.state.platform
        })
        return result

    @listen(generate_copywriting)
    def review_and_publish(self, copywriting):
        """审核和发布"""
        crew = Crew(
            agents=[self.editor],
            tasks=[self.review_task]
        )
        result = crew.kickoff(inputs={"copywriting": copywriting})
        return result

# 使用Flow
```

```python
# ai/graph_rag.py
from langchain_community.graphs import Neo4jGraph
from langchain_openai import ChatOpenAI
from qdrant_client import QdrantClient

class ProductGraphRAG:
    """产品知识图谱RAG系统"""

    def __init__(self):
        # Neo4j知识图谱
        self.graph = Neo4jGraph(
            url=settings.NEO4J_URI,
            username=settings.NEO4J_USER,
            password=settings.NEO4J_PASSWORD
        )

        # Qdrant向量库
        self.vector_client = QdrantClient(
            url=settings.QDRANT_URL
        )

        # LLM
        self.llm = ChatOpenAI(
            base_url="https://api.deepseek.com",
            api_key=settings.DEEPSEEK_API_KEY,
            model="deepseek-chat"
        )

    async def query(
        self,
        question: str,
        product_name: Optional[str] = None
    ) -> Dict:
        """混合检索查询"""

        # 1. 向量检索（语义相似）
        vector_results = await self._vector_search(question)

        # 2. 图遍历（关系推理）
        graph_results = await self._graph_traverse(
            question,
            product_name
        )

        # 3. 结果融合
        merged = self._merge_results(vector_results, graph_results)

        # 4. LLM生成答案
        answer = await self._generate_answer(question, merged)

        return {
            "answer": answer,
            "sources": merged,
            "confidence": self._calculate_confidence(merged)
        }

    async def _graph_traverse(
        self,
        question: str,
        product_name: str
    ):
        """图遍历查询"""
        # 根据问题生成Cypher查询
        cypher_template = """
        MATCH (p:Product {name: $product_name})
              -[r]->(related)
        RETURN type(r) as relationship,
               labels(related) as labels,
               related
        LIMIT 20
        """

        result = self.graph.query(
            cypher_template,
            {"product_name": product_name}
        )

        return result
```

```python
# plugins/base.py
from abc import ABC, abstractmethod

class Plugin(ABC):
    """插件基类"""

    @property
    @abstractmethod
    def name(self) -> str:
        pass

    @abstractmethod
    async def execute(
        self,
        inputs: Dict,
        context: PluginContext
    ) -> Dict:
        pass

# plugins/platform_adapter.py
class PlatformAdapterPlugin(Plugin):
    """平台适配插件"""

    name = "platform_adapter"

    ADAPTERS = {
        "wechat": WeChatAdapter(),
        "douyin": DouyinAdapter(),
        "web": WebAdapter()
    }

    async def execute(self, inputs: Dict, context: PluginContext):
        content = inputs["content"]
        platform = inputs["platform"]

        adapter = self.ADAPTERS.get(platform)
        if not adapter:
            raise ValueError(f"Unsupported platform: {platform}")

        adapted = await adapter.adapt(content)

        return {
            "adapted_content": adapted,
            "platform": platform
        }

# plugins/registry.py
class PluginRegistry:
    """插件注册表"""

    _instance = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
            cls._instance._plugins = {}
        return cls._instance

    def register(self, plugin: Plugin):
        self._plugins[plugin.name] = plugin

    def get(self, name: str) -> Plugin:
        return self._plugins.get(name)

    def list(self) -> List[str]:
        return list(self._plugins.keys())

# 使用示例
registry = PluginRegistry()
registry.register(PlatformAdapterPlugin())
registry.register(SEOOptimizerPlugin())
registry.register(ImageGeneratorPlugin())
```

```python
# core/quota.py
from enum import Enum

class SubscriptionTier(Enum):
    FREE = "free"          # 免费试用
    BASIC = "basic"        # 基础版
    PRO = "pro"            # 专业版
    ENTERPRISE = "enterprise"  # 企业版

class QuotaManager:
    """配额管理"""

    LIMITS = {
        SubscriptionTier.FREE: {
            "ai_calls_per_month": 50,
            "content_generations": 10,
            "storage_mb": 100,
        },
        SubscriptionTier.BASIC: {
            "ai_calls_per_month": 500,
            "content_generations": 100,
            "storage_mb": 1000,
        },
        SubscriptionTier.PRO: {
            "ai_calls_per_month": 5000,
            "content_generations": 1000,
            "storage_mb": 10000,
        },
        SubscriptionTier.ENTERPRISE: {
            "ai_calls_per_month": -1,  # 无限
            "content_generations": -1,
            "storage_mb": 100000,
        }
    }

    async def check_quota(
        self,
        tenant_id: str,
        resource: str
    ) -> bool:
        tenant = await self.get_tenant(tenant_id)
        usage = await self.get_usage(tenant_id, resource)
        limit = self.LIMITS[tenant.tier][resource]

        if limit == -1:
            return True

        return usage < limit

    async def consume_quota(
        self,
        tenant_id: str,
        resource: str,
        amount: int = 1
    ):
        if not await self.check_quota(tenant_id, resource):
            raise QuotaExceededException()

        await self.increment_usage(tenant_id, resource, amount)
```
