### 8.3 代码实现

```python
# backend/app/services/ip_agent/prompt_selector.py

from enum import Enum
from typing import Dict

class PromptVersion(Enum):
    SONNET_46 = "sonnet_46"      # 基线版本
    OPUS_47 = "opus_47"          # 自然度优先
    OPUS_48 = "opus_48"          # 实用性优先
    FABLE_5 = "fable_5"          # 创新性优先
    FUSION = "fusion"            # 融合版本（推荐）

class PromptSelector:
    """Prompt版本选择器"""
    
    @staticmethod
    def select_prompt(
        scenario: str = "production",
        user_type: str = "general",
        cost_sensitive: bool = False
    ) -> PromptVersion:
        """
        根据场景智能选择Prompt版本
        
        Args:
            scenario: 场景（production/mvp/demo）
            user_type: 用户类型（general/enterprise/vip）
            cost_sensitive: 是否成本敏感
        """
        if scenario == "demo":
            return PromptVersion.FABLE_5
        
        if cost_sensitive:
            return PromptVersion.OPUS_47
        
        if scenario == "mvp":
            return PromptVersion.OPUS_47
        
        if user_type == "vip":
            return PromptVersion.FUSION
        
        # 默认生产环境
        return PromptVersion.FUSION
```
